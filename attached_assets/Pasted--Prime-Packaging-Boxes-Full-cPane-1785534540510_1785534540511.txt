set -e

DB_USER="ecofrie1_primeuser"
DB_PASS="tumhara_db_password"
DB_NAME="ecofrie1_primedb"
SESSION_SECRET="prime-packaging-boxes-super-secret-key-2026-kawish-$(date +%s)"

echo "🧹 Cleaning public_html..."
rm -rf ~/public_html/* ~/public_html/.htaccess 2>/dev/null || true

echo "📥 Cloning repo..."
cd ~
rm -rf prime-packaging-boxes 2>/dev/null || true
git clone https://github.com/mohammadkawishbhatti-jpg/prime-packaging-boxes.git
cd prime-packaging-boxes

echo "📦 Installing pnpm..."
export PNPM_HOME="$HOME/.local/share/pnpm"
export PATH="$PNPM_HOME:$PATH"
curl -fsSL https://get.pnpm.io/install.sh | sh -
source ~/.bashrc 2>/dev/null || true
export PATH="$HOME/.local/share/pnpm:$PATH"

echo "📦 Installing dependencies..."
pnpm install --frozen-lockfile

echo "⚙️  Creating .env file..."
cat > ~/prime-packaging-boxes/.env << EOF
NODE_ENV=production
PORT=3001
DATABASE_URL=postgresql://${DB_USER}:${DB_PASS}@localhost:5432/${DB_NAME}
SESSION_SECRET=${SESSION_SECRET}
EOF

echo "🔨 Building main website..."
PORT=3001 BASE_PATH=/ pnpm --filter @workspace/prime-site run build

echo "🔨 Building admin panel..."
PORT=3001 BASE_PATH=/admin/ pnpm --filter @workspace/admin-panel run build

echo "🔨 Building customer portal..."
PORT=3001 BASE_PATH=/customer-portal/ pnpm --filter @workspace/customer-portal run build

echo "🔨 Building API server..."
pnpm --filter @workspace/api-server run build

echo "🗄️  Setting up database tables..."
pnpm --filter @workspace/db run push
